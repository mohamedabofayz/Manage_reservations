// --- 1. دالة إصلاح التواريخ (مهمة جداً لمنع الخطأ) --- function formatDateForInput(dateVal) { try { if (!dateVal) return ""; // تحويل القيمة إلى كائن تاريخ بغض النظر عن المصدر let d = new Date(dateVal); // التأكد من أن التاريخ صالح if (isNaN(d.getTime())) return ""; // إضافة يوم واحد إذا كان التوقيت يظهر كتاريخ سابق بسبب فرق التوقيت (UTC) // الحل الآمن لعرض التاريخ كما هو مسجل في جوجل شيت: // نستخدم toLocaleDateString مع تحايل بسيط لضمان صيغة YYYY-MM-DD const offset = d.getTimezoneOffset(); d = new Date(d.getTime() - (offset*60*1000)); return d.toISOString().split('T')[0]; } catch (e) { console.error("Error formatting date:", e); return ""; } } // --- 2. دالة ملء نموذج التعديل (المصححة بالكامل) --- function fillEditForm(invoiceData) { try { console.log("Starting fillEditForm with data:", invoiceData); // للمراقبة isEditMode = true; // تحديث العناوين والأزرار document.getElementById('formTitle').textContent = "تعديل الفاتورة: " + (invoiceData.invoice || "غير محدد"); document.getElementById('sendBtn').textContent = "تحديث الفاتورة"; document.getElementById('editModeAlert').style.display = "block"; // 1. ملء البيانات الأساسية document.getElementById('invoiceNumber').value = invoiceData.invoice || ''; document.getElementById('invoiceNumber').readOnly = true; document.querySelector('[name="clientName"]').value = invoiceData.clientName || ''; document.querySelector('[name="clientPhone"]').value = invoiceData.clientPhone || ''; document.querySelector('[name="address"]').value = invoiceData.address || ''; document.querySelector('[name="clientEmail"]').value = invoiceData.clientEmail || ''; document.querySelector('[name="hotelName"]').value = invoiceData.hotelName || ''; document.querySelector('[name="hotelPhone"]').value = invoiceData.hotelPhone || ''; document.querySelector('[name="hotelEmail"]').value = invoiceData.hotelEmail || ''; // إصلاح تاريخ الفاتورة document.querySelector('[name="todayDate"]').value = formatDateForInput(invoiceData.todayDate); // 2. إعادة تعيين حاوية الضيوف تماماً const container = document.getElementById('guestContainer'); container.innerHTML = ''; guestCount = 0; // تصفير العداد لضمان البدء من 1 // تفعيل زر الإضافة احتياطياً const addGuestBtn = document.getElementById('addGuestBtn'); addGuestBtn.disabled = false; addGuestBtn.textContent = '+ إضافة ضيف'; // 3. حلقة تكرار ذكية (تضمن عدم تخطي أي ضيف) // نبحث في البيانات من 1 إلى 4 for (let i = 1; i <= 4; i++) { // نتحقق هل يوجد اسم للضيف في هذا الرقم (i) في البيانات القادمة من السيرفر // أو هل يوجد سعر، لضمان أن هذا الصف يحتوي بيانات if (invoiceData[`name${i}`] || invoiceData[`amount${i}`] || invoiceData[`type${i}`]) { // نقوم بإضافة واجهة ضيف جديد (هذا سيجعل guestCount يزداد بـ 1) addGuest(); // الآن guestCount يمثل الرقم الحالي في النموذج (HTML) // و i يمثل الرقم في البيانات (Database) // يجب أن نستخدم guestCount لتحديد حقول الإدخال const form = document.getElementById('bookingForm'); // تعبئة الاسم let nameField = form.querySelector(`[name="name${guestCount}"]`); if (nameField) nameField.value = invoiceData[`name${i}`] || ''; // تعبئة التواريخ (مع الإصلاح) let checkinField = form.querySelector(`[name="checkin${guestCount}"]`); if (checkinField) checkinField.value = formatDateForInput(invoiceData[`checkin${i}`]); let checkoutField = form.querySelector(`[name="checkout${guestCount}"]`); if (checkoutField) checkoutField.value = formatDateForInput(invoiceData[`checkout${i}`]); // عدد الليالي let nightsField = form.querySelector(`[name="nights${guestCount}"]`); if (nightsField) nightsField.value = invoiceData[`nights${i}`] || ''; // --- تفكيك نوع الغرفة وعددها --- const rawType = invoiceData[`type${i}`] || ''; let roomTypeName = rawType; let roomCountVal = 1; // محاولة الفصل بناء على " X " if (rawType && typeof rawType === 'string' && rawType.includes(' X ')) { const parts = rawType.split(' X '); roomTypeName = parts[0].trim(); // نوع الغرفة // استخراج الرقم من الجزء الثاني const countMatch = parts[1].match(/\d+/); if (countMatch) { roomCountVal = parseInt(countMatch[0]); } } else if (rawType && typeof rawType === 'string') { // محاولة استخراج رقم إذا كان النص مثل "مزدوجة 2" const possibleNum = rawType.match(/(\d+)$/); if (possibleNum) { roomCountVal = parseInt(possibleNum[0]); roomTypeName = rawType.replace(possibleNum[0], '').trim(); } } // تعيين القيم في القائمة المنسدلة وحقل الرقم let typeSelect = form.querySelector(`[name="type${guestCount}"]`); if (typeSelect) typeSelect.value = roomTypeName; let roomsInput = form.querySelector(`[name="rooms${guestCount}"]`); if (roomsInput) roomsInput.value = roomCountVal; // السعر let amountField = form.querySelector(`[name="amount${guestCount}"]`); if (amountField) amountField.value = invoiceData[`amount${i}`] || ''; } } // إذا لم يكن هناك ضيوف، أضف واحداً فارغاً if (guestCount === 0) { addGuest(); } // تعبئة المجموع الكلي document.getElementById('totalField').value = invoiceData.total || '0.00'; // تمرير الشاشة للأعلى document.getElementById('appBox').scrollIntoView({ behavior: 'smooth' }); } catch (err) { console.error("Critical error in fillEditForm:", err); showMessage("حدث خطأ غير متوقع أثناء عرض البيانات: " + err.message, "error"); } }
